<script>
  const kladsSlider = document.getElementById('kladsSlider');
  const daysSlider = document.getElementById('daysSlider');
  const kladsValue = document.getElementById('kladsValue');
  const daysValue = document.getElementById('daysValue');
  const totalResult = document.getElementById('totalResult');
  const breakdown = document.getElementById('breakdown');

  const RATE = 1350;

  function formatNumber(num) {
    return num.toLocaleString('ru-RU');
  }

  function calculate() {
    const klads = parseInt(kladsSlider.value);
    const days = parseInt(daysSlider.value);
    const total = klads * days * RATE;

    kladsValue.textContent = klads;
    daysValue.textContent = days;
    totalResult.textContent = formatNumber(total);
    breakdown.textContent = `${klads} кладов × ${days} дней × ${formatNumber(RATE)} ₽`;
  }

  kladsSlider.addEventListener('input', calculate);
  daysSlider.addEventListener('input', calculate);
  calculate();
</script>
